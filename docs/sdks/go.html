Set up the client

This setup fragment reads credentials from the environment. Use the API operation reference to supply input for each call.

package main

import (
    "os"
    sdk "github.com/voxgig-sdk/lob-sdk/go"
)

func main() {
    client := sdk.NewLobSDK(map[string]any{"apikey": os.Getenv("LOB_APIKEY")})
    _ = client
}
